1. """Resource optimiser - monitors and optimises CPU/RAM usage."""
2. 
3. import logging
4. from typing import Any, Dict, Optional
5. 
6. 
7. logger = logging.getLogger(__name__)
8. 
9. 
10. class ResourceOptimizer:
11.     """Monitors system resource usage and applies optimisations."""
12. 
13.     def __init__(self, cpu_threshold: float = 85.0, ram_threshold: float = 90.0) -> None:
14.         self.cpu_threshold = cpu_threshold
15.         self.ram_threshold = ram_threshold
16. 
17.     # ------------------------------------------------------------------
18.     # Metrics
19.     # ------------------------------------------------------------------
20. 
21.     def get_metrics(self) -> Dict[str, Any]:
22.         """Return current CPU, RAM, and disk metrics."""
23.         try:
24.             import psutil  # type: ignore[import]
25. 
26.             cpu_pct = psutil.cpu_percent(interval=1)
27.             ram = psutil.virtual_memory()
28.             disk = psutil.disk_usage("/")
29.             net = psutil.net_io_counters()
30.             return {
31.                 "cpu_percent": cpu_pct,
32.                 "ram_percent": ram.percent,
33.                 "ram_used_gb": round(ram.used / 1024**3, 2),
34.                 "ram_total_gb": round(ram.total / 1024**3, 2),
35.                 "disk_percent": disk.percent,
36.                 "disk_used_gb": round(disk.used / 1024**3, 2),
37.                 "disk_total_gb": round(disk.total / 1024**3, 2),
38.                 "net_bytes_sent_mb": round(net.bytes_sent / 1024**2, 2),
39.                 "net_bytes_recv_mb": round(net.bytes_recv / 1024**2, 2),
40.             }
41.         except ImportError:
42.             logger.warning("psutil not installed; returning simulated metrics.")
43.             return self._simulate_metrics()
44. 
45.     def _simulate_metrics(self) -> Dict[str, Any]:
46.         import random
47. 
48.         return {
49.             "cpu_percent": round(random.uniform(10, 60), 1),
50.             "ram_percent": round(random.uniform(30, 70), 1),
51.             "ram_used_gb": round(random.uniform(1, 6), 2),
52.             "ram_total_gb": 8.0,
53.             "disk_percent": round(random.uniform(20, 50), 1),
54.             "disk_used_gb": round(random.uniform(20, 80), 2),
55.             "disk_total_gb": 160.0,
56.             "net_bytes_sent_mb": round(random.uniform(100, 5000), 2),
57.             "net_bytes_recv_mb": round(random.uniform(100, 5000), 2),
58.         }
59. 
60.     # ------------------------------------------------------------------
61.     # Analysis
62.     # ------------------------------------------------------------------
63. 
64.     def analyse(self) -> Dict[str, Any]:
65.         """Analyse current resource usage and return recommendations."""
66.         metrics = self.get_metrics()
67.         recommendations = []
68. 
69.         if metrics["cpu_percent"] > self.cpu_threshold:
70.             recommendations.append(
71.                 f"CPU usage is high ({metrics['cpu_percent']}%). Consider scaling up or optimising workloads."
72.             )
73.         if metrics["ram_percent"] > self.ram_threshold:
74.             recommendations.append(
75.                 f"RAM usage is high ({metrics['ram_percent']}%). Consider adding swap or scaling RAM."
76.             )
77.         if metrics["disk_percent"] > 80:
78.             recommendations.append(
79.                 f"Disk usage is high ({metrics['disk_percent']}%). Run cleanup or expand storage."
80.             )
81. 
82.         return {
83.             "metrics": metrics,
84.             "status": "warning" if recommendations else "healthy",
85.             "recommendations": recommendations,
86.         }
87. 
88.     def is_under_pressure(self) -> bool:
89.         """Return True if the system is under resource pressure."""
90.         metrics = self.get_metrics()
91.         return metrics["cpu_percent"] > self.cpu_threshold or metrics["ram_percent"] > self.ram_threshold
