1. """Data processing service — transforms and analyzes data payloads."""
2. import logging
3. import json
4. from typing import Any, Dict, List
5. 
6. logger = logging.getLogger(__name__)
7. 
8. 
9. class DataProcessor:
10.     """Processes data payloads with various operations."""
11. 
12.     SUPPORTED_OPERATIONS = [
13.         "summarize", "classify", "extract", "transform",
14.         "validate", "aggregate", "filter", "sort",
15.     ]
16. 
17.     async def process(
18.         self,
19.         data: Any,
20.         operation: str,
21.         options: Dict[str, Any] = None,
22.     ) -> Dict[str, Any]:
23.         """Process data with the given operation."""
24.         if operation not in self.SUPPORTED_OPERATIONS:
25.             return {
26.                 "status": "error",
27.                 "message": f"Unsupported operation: {operation}. "
28.                            f"Supported: {self.SUPPORTED_OPERATIONS}",
29.             }
30. 
31.         options = options or {}
32.         handler = getattr(self, f"_op_{operation}", self._op_default)
33.         result = await handler(data, options)
34. 
35.         logger.info("DataProcessor: op=%s input_type=%s", operation, type(data).__name__)
36.         return {
37.             "status": "processed",
38.             "operation": operation,
39.             "input_type": type(data).__name__,
40.             "result": result,
41.         }
42. 
43.     async def _op_summarize(self, data: Any, opts: Dict) -> str:
44.         if isinstance(data, dict):
45.             return f"Dict with {len(data)} keys: {list(data.keys())[:5]}"
46.         if isinstance(data, list):
47.             return f"List with {len(data)} items"
48.         return str(data)[:200]
49. 
50.     async def _op_classify(self, data: Any, opts: Dict) -> Dict:
51.         return {"type": type(data).__name__, "size": len(str(data))}
52. 
53.     async def _op_extract(self, data: Any, opts: Dict) -> Any:
54.         fields = opts.get("fields", [])
55.         if isinstance(data, dict) and fields:
56.             return {k: data.get(k) for k in fields}
57.         return data
58. 
59.     async def _op_transform(self, data: Any, opts: Dict) -> Any:
60.         fmt = opts.get("format", "json")
61.         if fmt == "json":
62.             return json.loads(json.dumps(data, default=str))
63.         return str(data)
64. 
65.     async def _op_validate(self, data: Any, opts: Dict) -> Dict:
66.         schema = opts.get("schema", {})
67.         errors = []
68.         if schema and isinstance(data, dict):
69.             for field, typ in schema.items():
70.                 if field not in data:
71.                     errors.append(f"Missing field: {field}")
72.         return {"valid": len(errors) == 0, "errors": errors}
73. 
74.     async def _op_aggregate(self, data: Any, opts: Dict) -> Any:
75.         if isinstance(data, list) and all(isinstance(x, (int, float)) for x in data):
76.             return {
77.                 "count": len(data), "sum": sum(data),
78.                 "min": min(data), "max": max(data),
79.                 "avg": sum(data) / len(data) if data else 0,
80.             }
81.         return {"count": len(data) if hasattr(data, "__len__") else 1}
82. 
83.     async def _op_filter(self, data: Any, opts: Dict) -> Any:
84.         condition = opts.get("condition", {})
85.         if isinstance(data, list) and condition:
86.             key = condition.get("key")
87.             value = condition.get("value")
88.             if key:
89.                 return [item for item in data if isinstance(item, dict) and item.get(key) == value]
90.         return data
91. 
92.     async def _op_sort(self, data: Any, opts: Dict) -> Any:
93.         if isinstance(data, list):
94.             key = opts.get("key")
95.             reverse = opts.get("reverse", False)
96.             if key:
97.                 return sorted(data, key=lambda x: x.get(key, 0) if isinstance(x, dict) else x, reverse=reverse)
98.             return sorted(data, reverse=reverse)
99.         return data
100. 
101.     async def _op_default(self, data: Any, opts: Dict) -> Any:
102.         return data
