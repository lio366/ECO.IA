35.     def clean_old_logs(self) -> Dict[str, Any]:
36.         """Remove log files older than *max_log_age_days*."""
37.         cutoff = datetime.utcnow() - timedelta(days=self.max_log_age_days)
38.         removed = []
39.         freed_bytes = 0
40. 
41.         for log_dir in self.log_dirs:
42.             if not Path(log_dir).exists():
43.                 continue
44.             for file_path in Path(log_dir).rglob("*.log"):
45.                 try:
46.                     mtime = datetime.utcfromtimestamp(file_path.stat().st_mtime)
47.                     if mtime < cutoff:
48.                         size = file_path.stat().st_size
49.                         file_path.unlink()
50.                         freed_bytes += size
