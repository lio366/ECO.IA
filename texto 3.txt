1. """Predictor - simple time-series anomaly detection."""
2. 
3. import logging
4. import statistics
5. from collections import deque
6. from typing import Any, Deque, Dict, List, Optional, Tuple
7. 
8. 
9. logger = logging.getLogger(__name__)
10. 
11. 
12. class Predictor:
13.     """Lightweight anomaly predictor using rolling statistics."""
14. 
15.     def __init__(self, window_size: int = 20, z_score_threshold: float = 2.5) -> None:
16.         self.window_size = window_size
17.         self.z_score_threshold = z_score_threshold
18.         self._series: Dict[str, Deque[float]] = {}
19.         self._predictions: List[Dict[str, Any]] = []
20. 
21.     # ------------------------------------------------------------------
22.     # Data ingestion
23.     # ------------------------------------------------------------------
24. 
25.     def record(self, metric: str, value: float) -> None:
26.         """Add a new observation for *metric*."""
27.         if metric not in self._series:
28.             self._series[metric] = deque(maxlen=self.window_size)
29.         self._series[metric].append(value)
30. 
31.     # ------------------------------------------------------------------
32.     # Anomaly detection
33.     # ------------------------------------------------------------------
34. 
35.     def is_anomaly(self, metric: str, value: float) -> Tuple[bool, Optional[float]]:
36.         """Return (is_anomaly, z_score) for the given value."""
37.         series = list(self._series.get(metric, []))
38.         if len(series) < 3:
39.             return False, None
40.         try:
41.             mean = statistics.mean(series)
42.             std = statistics.stdev(series)
43.         except statistics.StatisticsError:
44.             return False, None
45. 
46.         if std == 0:
47.             return False, 0.0
48. 
49.         z_score = abs(value - mean) / std
50.         return z_score > self.z_score_threshold, round(z_score, 3)
