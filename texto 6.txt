1. #!/usr/bin/env bash
2. # =============================================================================
3. # ECO-IA – Automated Backup Script
4. # =============================================================================
5. set -euo pipefail
6. 
7. # Load environment
8. ECO_IA_DIR="${ECO_IA_DIR:-/opt/eco-ia}"
9. # shellcheck source=/dev/null
10. [[ -f "$ECO_IA_DIR/.env" ]] && source "$ECO_IA_DIR/.env"
11. 
12. TIMESTAMP=$(date +"%Y%m%d_%H%M%S")
13. BACKUP_NAME="eco-ia-backup-$TIMESTAMP"
14. REMOTE_PATH="${HETZNER_STORAGE_BOX_USER:-}@${HETZNER_STORAGE_BOX_HOST:-}:/backups/eco-ia"
15. RETENTION_DAYS="${BACKUP_RETENTION_DAYS:-30}"
16. LOCAL_BACKUP_DIR="${LOCAL_BACKUP_DIR:-}"
17. if [[ -z "$LOCAL_BACKUP_DIR" ]]; then
18.     LOCAL_BACKUP_DIR=$(mktemp -d -t eco-ia-backups-XXXXXX)
19.     CLEANUP_TMPDIR=1
20. else
21.     mkdir -p "$LOCAL_BACKUP_DIR"
22.     CLEANUP_TMPDIR=0
23. fi
24. 
25. echo "[$TIMESTAMP] Starting ECO-IA backup…"
26. 
27. # ── 1. PostgreSQL dump ───────────────────────────────────────────────────────
28. echo "Dumping PostgreSQL database…"
29. docker exec eco-ia-postgres pg_dump -U eco_ia eco_ia \
30.     | gzip > "$LOCAL_BACKUP_DIR/${BACKUP_NAME}-db.sql.gz" \
31.     && echo "✅ Database dump complete." \
32.     || echo "❌ Database dump FAILED."
33. 
34. # ── 2. Configuration backup ──────────────────────────────────────────────────
35. echo "Backing up configuration…"
36. tar -czf "$LOCAL_BACKUP_DIR/${BACKUP_NAME}-config.tar.gz" \
37.     "$ECO_IA_DIR/.env" \
38.     "$ECO_IA_DIR/config/" \
39.     2>/dev/null \
40.     && echo "✅ Config backup complete." \
41.     || echo "❌ Config backup FAILED."
42. 
43. # ── 3. Upload to Hetzner Storage Box ─────────────────────────────────────────
44. if [[ -n "${HETZNER_STORAGE_BOX_HOST:-}" && -n "${HETZNER_STORAGE_BOX_USER:-}" ]]; then
45.     echo "Uploading to Hetzner Storage Box…"
46.     rsync -az --delete \
47.         -e "ssh -o StrictHostKeyChecking=no -p 23" \
48.         "$LOCAL_BACKUP_DIR/" \
49.         "$REMOTE_PATH/" \
50.         && echo "✅ Upload complete." \
51.         || echo "❌ Upload FAILED."
52. else
53.     echo "⚠  Hetzner Storage Box not configured; keeping local backup only."
54. fi
55. 
56. # ── 4. Clean old local backups ────────────────────────────────────────────────
57. echo "Cleaning backups older than $RETENTION_DAYS days…"
58. find "$LOCAL_BACKUP_DIR" -name "eco-ia-backup-*" -mtime "+$RETENTION_DAYS" -delete
59. echo "✅ Cleanup complete."
60. 
61. echo "[$TIMESTAMP] Backup finished: $BACKUP_NAME"
62. 
63. # ── 5. Clean up temporary directory if created by mktemp ──────────────────────
64. if [[ "$CLEANUP_TMPDIR" -eq 1 ]]; then
65.     rm -rf "$LOCAL_BACKUP_DIR"
66. fi
67.
