1. """Cleaner - automatic removal of logs and temporary files."""
2. 
3. import glob
4. import logging
5. import os
6. import shutil
7. from datetime import datetime, timedelta
8. from pathlib import Path
9. from typing import Any, Dict, List, Optional
10. 
11. 
12. logger = logging.getLogger(__name__)
13. 
14. 
15. class Cleaner:
16.     """Automatically cleans up old logs, temp files, and Docker artefacts."""
17. 
18.     def __init__(
19.         self,
20.         log_dirs: Optional[List[str]] = None,
21.         temp_dirs: Optional[List[str]] = None,
22.         max_log_age_days: int = 7,
23.         max_log_size_mb: float = 100.0,
24.     ) -> None:
25.         self.log_dirs = log_dirs or ["/var/log/eco-ia", "/opt/eco-ia/logs"]
26.         self.temp_dirs = temp_dirs or []  # caller must supply safe temp dirs explicitly
27.         self.max_log_age_days = max_log_age_days
28.         self.max_log_size_mb = max_log_size_mb
29.         self._cleanup_log: List[Dict[str, Any]] = []
30. 
31.     # ------------------------------------------------------------------
32.     # Clean operations
33.     # ------------------------------------------------------------------
34. 
35.     def clean_old_logs(self) -> Dict[str, Any]:
36.         """Remove log files older than *max_log_age_days*."""
37.         cutoff = datetime.utcnow() - timedelta(days=self.max_log_age_days)
38.         removed = []
39.         freed_bytes = 0
40. 
41.         for log_dir in self.log_dirs:
42.             if not Path(log_dir).exists():
43.                 continue
44.             for file_path in Path(log_dir).rglob("*.log"):
45.                 try:
46.                     mtime = datetime.utcfromtimestamp(file_path.stat().st_mtime)
47.                     if mtime < cutoff:
48.                         size = file_path.stat().st_size
49.                         file_path.unlink()
50.                         freed_bytes += size
51.                         removed.append(str(file_path))
52.                 except OSError as exc:
53.                     logger.warning("Could not remove '%s': %s", file_path, exc)
54. 
55.         result: Dict[str, Any] = {
56.             "action": "clean_old_logs",
57.             "removed_count": len(removed),
58.             "freed_mb": round(freed_bytes / 1024**2, 2),
59.             "timestamp": datetime.utcnow().isoformat(),
60.         }
61.         self._cleanup_log.append(result)
62.         logger.info("Cleaned %d old log files (%.2f MB freed).", len(removed), result["freed_mb"])
63.         return result
64. 
65.     def clean_temp_files(self) -> Dict[str, Any]:
66.         """Remove temporary directories."""
67.         removed = []
68.         freed_bytes = 0
69. 
70.         for tmp_dir in self.temp_dirs:
71.             tmp_path = Path(tmp_dir)
72.             if not tmp_path.exists():
73.                 continue
74.             try:
75.                 size = sum(f.stat().st_size for f in tmp_path.rglob("*") if f.is_file())
76.                 shutil.rmtree(tmp_path, ignore_errors=True)
77.                 freed_bytes += size
78.                 removed.append(str(tmp_path))
79.             except OSError as exc:
80.                 logger.warning("Could not clean '%s': %s", tmp_path, exc)
81. 
82.         result: Dict[str, Any] = {
83.             "action": "clean_temp_files",
84.             "cleaned_dirs": removed,
85.             "freed_mb": round(freed_bytes / 1024**2, 2),
86.             "timestamp": datetime.utcnow().isoformat(),
87.         }
88.         self._cleanup_log.append(result)
89.         logger.info("Cleaned temp files (%.2f MB freed).", result["freed_mb"])
90.         return result
91. 
92.     def clean_docker_artefacts(self) -> Dict[str, Any]:
93.         """Remove unused Docker images, containers, and volumes."""
94.         import subprocess
95. 
96.         commands = [
97.             ["docker", "container", "prune", "-f"],
98.             ["docker", "image", "prune", "-f"],
99.             ["docker", "volume", "prune", "-f"],
100.             ["docker", "network", "prune", "-f"],
101.         ]
102.         results = []
103.         for cmd in commands:
104.             proc = subprocess.run(cmd, capture_output=True, text=True, check=False)  # noqa: S603
105.             results.append({
106.                 "cmd": " ".join(cmd),
107.                 "success": proc.returncode == 0,
108.                 "output": proc.stdout[-200:] if proc.stdout else "",
109.             })
110. 
111.         result: Dict[str, Any] = {
112.             "action": "clean_docker_artefacts",
113.             "results": results,
114.             "timestamp": datetime.utcnow().isoformat(),
115.         }
116.         self._cleanup_log.append(result)
117.         return result
118. 
119.     def run_all(self) -> List[Dict[str, Any]]:
120.         """Run all cleanup tasks."""
121.         return [
122.             self.clean_old_logs(),
123.             self.clean_temp_files(),
124.             self.clean_docker_artefacts(),
125.         ]
126. 
127.     def get_cleanup_history(self, limit: int = 20) -> List[Dict[str, Any]]:
128.         return self._cleanup_log[-limit:]
