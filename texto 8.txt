34. 
35. class PricingEngine:
36.     """Computes dynamic pricing based on demand and resource usage."""
37. 
38.     def __init__(self, demand_multiplier: float = 1.0) -> None:
39.         self._demand_multiplier = demand_multiplier
40.         self._plans = dict(PLANS)
41. 
42.     # ------------------------------------------------------------------
43.     # Price calculation
44.     # ------------------------------------------------------------------
45. 
46.     def get_price(self, plan: str, usage_pct: float = 0.0) -> float:
47.         """Calculate the current price for *plan* considering demand.
48. 
49.         Args:
50.             plan: Plan name ("basic", "pro", "enterprise").
51.             usage_pct: Current server utilisation percentage (0–100).
52. 
53.         Returns:
54.             Adjusted price in USD.
55.         """
56.         plan_config = self._plans.get(plan)
57.         if plan_config is None:
58.             raise ValueError(f"Unknown plan: {plan!r}")
59. 
60.         base = plan_config["base_price_usd"]
61.         demand_factor = self._compute_demand_factor(usage_pct)
62.         price = round(base * self._demand_multiplier * demand_factor, 2)
63.         logger.debug("Price for '%s': base=%.2f demand_factor=%.2f → %.2f", plan, base, demand_factor, price)
64.         return price
65. 
66.     def _compute_demand_factor(self, usage_pct: float) -> float:
67.         """Return a multiplier based on server utilisation."""
68.         if usage_pct < 50:
69.             return 1.0          # normal pricing
70.         if usage_pct < 75:
71.             return 1.1          # slight surge
72.         if usage_pct < 90:
73.             return 1.25         # surge pricing
74.         return 1.5              # peak pricing
75. 
76.     # ------------------------------------------------------------------
77.     # Plan management
78.     # ------------------------------------------------------------------
79. 
80.     def list_plans(self) -> List[Dict[str, Any]]:
81.         """Return all available plans with current prices."""
82.         return [
83.             {**v, "plan_key": k, "current_price_usd": self.get_price(k)}
84.             for k, v in self._plans.items()
85.         ]
86. 
87.     def recommend_plan(self, monthly_api_calls: int, storage_gb: int) -> str:
88.         """Recommend the cheapest plan that covers the given requirements."""
89.         for plan_key in ("basic", "pro", "enterprise"):
90.             plan = self._plans[plan_key]
91.             if (
92.                 plan["api_calls_per_month"] >= monthly_api_calls
93.                 and plan["storage_gb"] >= storage_gb
94.             ):
95.                 return plan_key
96.         return "enterprise"
97. 
98.     def set_demand_multiplier(self, multiplier: float) -> None:
99.         """Override the global demand multiplier (1.0 = normal)."""
100.         if multiplier <= 0:
101.             raise ValueError("Demand multiplier must be positive.")
102.         self._demand_multiplier = multiplier
103.         logger.info("Demand multiplier set to %.2f", multiplier)
104. 
105.     def get_pricing_summary(self) -> Dict[str, Any]:
106.         return {
107.             "demand_multiplier": self._demand_multiplier,
108.             "plans": self.list_plans(),
109.         }
