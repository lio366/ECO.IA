1. """Reporter - generates automated reports and sends email alerts."""
2. 
3. import logging
4. import smtplib
5. from datetime import datetime
6. from email.mime.multipart import MIMEMultipart
7. from email.mime.text import MIMEText
8. from typing import Any, Dict, List, Optional
9. 
10. 
11. logger = logging.getLogger(__name__)
12. 
13. 
14. class Reporter:
15.     """Generates and distributes automated system reports."""
16. 
17.     def __init__(
18.         self,
19.         smtp_host: Optional[str] = None,
20.         smtp_port: int = 587,
21.         smtp_user: Optional[str] = None,
22.         smtp_password: Optional[str] = None,
23.         from_email: Optional[str] = None,
24.         to_emails: Optional[List[str]] = None,
25.     ) -> None:
26.         self.smtp_host = smtp_host
27.         self.smtp_port = smtp_port
28.         self.smtp_user = smtp_user
29.         self.smtp_password = smtp_password
30.         self.from_email = from_email or smtp_user
31.         self.to_emails = to_emails or []
32.         self._reports: List[Dict[str, Any]] = []
33. 
34.     # ------------------------------------------------------------------
35.     # Report generation
36.     # ------------------------------------------------------------------
37. 
38.     def generate_daily_report(self, data: Dict[str, Any]) -> Dict[str, Any]:
39.         """Generate a structured daily report from *data*."""
40.         report = {
41.             "type": "daily",
42.             "generated_at": datetime.utcnow().isoformat(),
43.             "summary": self._build_summary(data),
44.             "data": data,
45.         }
46.         self._reports.append(report)
47.         logger.info("Daily report generated.")
48.         return report
49. 
50.     def generate_alert_report(self, alert: Dict[str, Any]) -> Dict[str, Any]:
51.         """Generate an alert report for a critical event."""
52.         report = {
53.             "type": "alert",
54.             "generated_at": datetime.utcnow().isoformat(),
55.             "alert": alert,
56.         }
57.         self._reports.append(report)
58.         logger.warning("Alert report generated: %s", alert.get("message"))
59.         return report
60. 
61.     def _build_summary(self, data: Dict[str, Any]) -> str:
62.         lines = ["ECO-IA Daily Report", "=" * 40]
63.         for key, value in data.items():
64.             lines.append(f"{key}: {value}")
65.         return "\n".join(lines)
66. 
67.     # ------------------------------------------------------------------
68.     # Email delivery
69.     # ------------------------------------------------------------------
70. 
71.     def send_email(self, subject: str, body: str, to: Optional[List[str]] = None) -> bool:
72.         """Send an email report. Returns True on success."""
73.         recipients = to or self.to_emails
74.         if not recipients or not self.smtp_host:
75.             logger.warning("Email not configured; skipping send.")
76.             return False
77. 
78.         try:
79.             msg = MIMEMultipart("alternative")
80.             msg["Subject"] = subject
81.             msg["From"] = self.from_email or ""
82.             msg["To"] = ", ".join(recipients)
83.             msg.attach(MIMEText(body, "plain", "utf-8"))
84. 
85.             with smtplib.SMTP(self.smtp_host, self.smtp_port) as server:
86.                 server.ehlo()
87.                 server.starttls()
88.                 if self.smtp_user and self.smtp_password:
89.                     server.login(self.smtp_user, self.smtp_password)
90.                 server.sendmail(self.from_email or "", recipients, msg.as_string())
91. 
92.             logger.info("Email sent to %s.", recipients)
93.             return True
94.         except Exception as exc:  # noqa: BLE001
95.             logger.error("Failed to send email: %s", exc)
96.             return False
97. 
98.     def send_daily_report_email(self, data: Dict[str, Any]) -> bool:
99.         report = self.generate_daily_report(data)
100.         subject = f"ECO-IA Daily Report – {datetime.utcnow().strftime('%Y-%m-%d')}"
101.         return self.send_email(subject, report["summary"])
102. 
103.     def get_reports(self, report_type: Optional[str] = None, limit: int = 20) -> List[Dict[str, Any]]:
104.         reports = self._reports
105.         if report_type:
106.             reports = [r for r in reports if r.get("type") == report_type]
107.         return reports[-limit:]
