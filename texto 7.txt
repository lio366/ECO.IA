1. """Client management for the monetization agent."""
2. 
3. import logging
4. import uuid
5. from datetime import datetime
6. from typing import Any, Dict, List, Optional
7. 
8. 
9. logger = logging.getLogger(__name__)
10. 
11. 
12. class Client:
13.     """Represents a paying client of the ECO-IA system."""
14. 
15.     def __init__(
16.         self,
17.         client_id: str,
18.         name: str,
19.         email: str,
20.         plan: str = "basic",
21.         stripe_customer_id: Optional[str] = None,
22.     ) -> None:
23.         self.client_id = client_id
24.         self.name = name
25.         self.email = email
26.         self.plan = plan
27.         self.stripe_customer_id = stripe_customer_id
28.         self.created_at = datetime.utcnow()
29.         self.last_activity: Optional[datetime] = None
30.         self.is_active = True
31.         self.monthly_spend: float = 0.0
32.         self.upsell_opportunities: List[str] = []
33. 
34.     def to_dict(self) -> Dict[str, Any]:
35.         return {
36.             "client_id": self.client_id,
37.             "name": self.name,
38.             "email": self.email,
39.             "plan": self.plan,
40.             "stripe_customer_id": self.stripe_customer_id,
41.             "created_at": self.created_at.isoformat(),
42.             "last_activity": self.last_activity.isoformat() if self.last_activity else None,
43.             "is_active": self.is_active,
44.             "monthly_spend": self.monthly_spend,
45.             "upsell_opportunities": self.upsell_opportunities,
46.         }
47. 
48. 
49. class ClientManager:
50.     """Manages the lifecycle of ECO-IA clients."""
51. 
52.     def __init__(self) -> None:
53.         self._clients: Dict[str, Client] = {}
54. 
55.     # ------------------------------------------------------------------
56.     # CRUD
57.     # ------------------------------------------------------------------
58. 
59.     def create_client(
60.         self,
61.         name: str,
62.         email: str,
63.         plan: str = "basic",
64.         stripe_customer_id: Optional[str] = None,
65.     ) -> Client:
66.         client_id = str(uuid.uuid4())
67.         client = Client(
68.             client_id=client_id,
69.             name=name,
70.             email=email,
71.             plan=plan,
72.             stripe_customer_id=stripe_customer_id,
73.         )
74.         self._clients[client_id] = client
75.         logger.info("Client created: %s (%s)", name, client_id)
76.         return client
77. 
78.     def get_client(self, client_id: str) -> Optional[Client]:
79.         return self._clients.get(client_id)
80. 
81.     def list_clients(self, active_only: bool = True) -> List[Client]:
82.         if active_only:
83.             return [c for c in self._clients.values() if c.is_active]
84.         return list(self._clients.values())
85. 
86.     def deactivate_client(self, client_id: str) -> bool:
87.         client = self._clients.get(client_id)
88.         if client:
89.             client.is_active = False
90.             logger.info("Client deactivated: %s", client_id)
91.             return True
92.         return False
93. 
94.     def upgrade_plan(self, client_id: str, new_plan: str) -> bool:
95.         client = self._clients.get(client_id)
96.         if client:
97.             old_plan = client.plan
98.             client.plan = new_plan
99.             logger.info("Client '%s' upgraded from '%s' to '%s'.", client_id, old_plan, new_plan)
100.             return True
101.         return False
102. 
103.     # ------------------------------------------------------------------
104.     # Analytics
105.     # ------------------------------------------------------------------
106. 
107.     def record_activity(self, client_id: str) -> None:
108.         client = self._clients.get(client_id)
109.         if client:
110.             client.last_activity = datetime.utcnow()
111. 
112.     def detect_upsell_opportunities(self) -> List[Dict[str, Any]]:
113.         """Identify clients that could benefit from a plan upgrade."""
114.         opportunities = []
115.         for client in self._clients.values():
116.             if not client.is_active:
117.                 continue
118.             if client.plan == "basic" and client.monthly_spend > 50:
119.                 opportunities.append(
120.                     {
121.                         "client_id": client.client_id,
122.                         "name": client.name,
123.                         "current_plan": client.plan,
124.                         "suggested_plan": "pro",
125.                         "reason": "High monthly spend on basic plan",
126.                     }
127.                 )
128.             elif client.plan == "pro" and client.monthly_spend > 200:
129.                 opportunities.append(
130.                     {
131.                         "client_id": client.client_id,
132.                         "name": client.name,
133.                         "current_plan": client.plan,
134.                         "suggested_plan": "enterprise",
135.                         "reason": "High monthly spend on pro plan",
136.                     }
137.                 )
138.         return opportunities
139. 
140.     def get_summary(self) -> Dict[str, Any]:
141.         active = self.list_clients(active_only=True)
142.         return {
143.             "total_clients": len(self._clients),
144.             "active_clients": len(active),
145.             "plans": {plan: sum(1 for c in active if c.plan == plan) for plan in {"basic", "pro", "enterprise"}},
146.         }
