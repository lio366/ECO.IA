1. """Dashboard data aggregator for the analytics agent."""
2. 
3. import logging
4. from datetime import datetime
5. from typing import Any, Dict, List, Optional
6. 
7. 
8. logger = logging.getLogger(__name__)
9. 
10. 
11. class DashboardData:
12.     """Aggregates metrics from all agents for the Grafana/web dashboard."""
13. 
14.     def __init__(self) -> None:
15.         self._snapshots: List[Dict[str, Any]] = []
16. 
17.     def snapshot(
18.         self,
19.         resources: Optional[Dict[str, Any]] = None,
20.         monetization: Optional[Dict[str, Any]] = None,
21.         security: Optional[Dict[str, Any]] = None,
22.         devops: Optional[Dict[str, Any]] = None,
23.     ) -> Dict[str, Any]:
24.         """Capture a dashboard snapshot from agent data."""
25.         snap = {
26.             "timestamp": datetime.utcnow().isoformat(),
27.             "resources": resources or {},
28.             "monetization": monetization or {},
29.             "security": security or {},
30.             "devops": devops or {},
31.         }
32.         self._snapshots.append(snap)
33.         if len(self._snapshots) > 1440:  # keep ~24h at 1-min intervals
34.             self._snapshots = self._snapshots[-1440:]
35.         return snap
36. 
37.     def get_latest(self) -> Optional[Dict[str, Any]]:
38.         return self._snapshots[-1] if self._snapshots else None
39. 
40.     def get_history(self, limit: int = 60) -> List[Dict[str, Any]]:
41.         return self._snapshots[-limit:]
42. 
43.     def get_kpis(self) -> Dict[str, Any]:
44.         """Compute high-level KPIs from the latest snapshot."""
45.         latest = self.get_latest()
46.         if not latest:
47.             return {"status": "no_data"}
48. 
49.         mon = latest.get("monetization", {})
50.         res = latest.get("resources", {})
51.         sec = latest.get("security", {})
52. 
53.         return {
54.             "timestamp": latest["timestamp"],
55.             "active_clients": mon.get("active_clients", 0),
56.             "monthly_revenue_usd": mon.get("monthly_revenue_usd", 0.0),
57.             "cpu_percent": res.get("cpu_percent", 0.0),
58.             "ram_percent": res.get("ram_percent", 0.0),
59.             "disk_percent": res.get("disk_percent", 0.0),
60.             "security_alerts": sec.get("total_alerts", 0),
61.             "system_health": self._compute_health(res, sec),
62.         }
63. 
64.     def _compute_health(self, resources: Dict[str, Any], security: Dict[str, Any]) -> str:
65.         if security.get("high_severity", 0) > 0:
66.             return "critical"
67.         cpu = resources.get("cpu_percent", 0)
68.         ram = resources.get("ram_percent", 0)
69.         if cpu > 90 or ram > 90:
70.             return "warning"
71.         return "healthy"
