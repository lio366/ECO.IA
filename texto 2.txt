22.     def check_open_ports(self) -> Dict[str, Any]:
23.         """List open ports using ss."""
24.         result = subprocess.run(  # noqa: S603
25.             ["ss", "-tlnp"],
26.             capture_output=True,
27.             text=True,
28.             check=False,
29.         )
30.         return {
31.             "check": "open_ports",
32.             "output": result.stdout,
33.             "success": result.returncode == 0,
34.             "timestamp": datetime.utcnow().isoformat(),
35.         }
36.

39.         result = subprocess.run(  # noqa: S603

43.             check=False,
44.         )
