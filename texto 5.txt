38. class TestMessageBus:
39.     def setup_method(self):
40.         self.bus = MessageBus()
41. 
42.     @pytest.mark.asyncio
43.     async def test_publish_and_subscribe(self):
44.         received = []
45. 
46.         async def handler(msg: Message):
47.             received.append(msg)
48. 
49.         await self.bus.subscribe("agent_a", handler)
50.         msg = Message(sender="agent_b", target="agent_a", content={"type": "ping"})
51.         await self.bus.publish(msg)
52. 
53.         assert len(received) == 1
54.         assert received[0].content["type"] == "ping"
55. 
56.     @pytest.mark.asyncio
57.     async def test_broadcast(self):
58.         received = []
59. 
60.         async def handler(msg: Message):
61.             received.append(msg)
62. 
63.         await self.bus.subscribe("agent_a", handler)
64.         msg = Message(sender="orchestrator", target="*", content={"type": "broadcast"})
65.         await self.bus.publish(msg)
66. 
67.         assert len(received) == 1
68. 
69.     def test_get_stats(self):
70.         stats = self.bus.get_stats()
71.         assert "total_messages" in stats
72.         assert "subscribers" in stats
73. 
74.     def test_get_history(self):
75.         history = self.bus.get_history()
76.         assert isinstance(history, list)

353. 
354.     def test_record_and_predict(self):
355.         for v in [1.0, 1.1, 1.0, 0.9, 1.0, 1.1, 1.0, 0.9]:
356.             self.predictor.record("cpu", v)
357.         prediction = self.predictor.predict_next("cpu")
358.         assert prediction is not None
359. 
360.     def test_no_anomaly_normal_values(self):
361.         for v in [50.0, 51.0, 50.5, 49.5, 50.0, 51.0]:
362.             self.predictor.record("ram", v)
363.         anomaly, _ = self.predictor.is_anomaly("ram", 50.5)
364.         assert anomaly is False
365. 
366.     def test_anomaly_detected(self):
367.         for v in [10.0, 11.0, 9.5, 10.5, 10.0, 11.0, 9.5, 10.5]:
368.             self.predictor.record("metric", v)
369.         anomaly, z_score = self.predictor.is_anomaly("metric", 100.0)
370.         assert anomaly is True
371.         assert z_score is not None and z_score > 2.0
372. 
373.     def test_check_and_alert(self):
374.         for v in [5.0] * 8:
375.             self.predictor.record("disk", v)
376.         alert = self.predictor.check_and_alert("disk", 999.0)
377.         assert alert is not None
378.         assert alert["metric"] == "disk"
379. 
380.     def test_get_series_stats(self):
381.         self.predictor.record("temp", 42.0)
382.         stats = self.predictor.get_series_stats("temp")
383.         assert stats["count"] == 1
