1. """Task scheduler for ECO-IA agents."""
2. import asyncio, logging
3. from datetime import datetime
4. from typing import Any, Callable, Coroutine, Dict, List, Optional
5. logger = logging.getLogger(__name__)
6. 
7. class ScheduledTask:
8.     def __init__(self, task_id, func, interval_seconds, description=""):
9.         self.task_id = task_id; self.func = func
10.         self.interval_seconds = interval_seconds; self.description = description
11.         self.last_run = None; self.run_count = 0; self.error_count = 0; self._task = None
12. 
13. class TaskScheduler:
14.     def __init__(self): self._tasks: Dict[str, ScheduledTask] = {}; self._running = False
15.     def register(self, task_id, func, interval_seconds, description=""):
16.         t = ScheduledTask(task_id, func, interval_seconds, description)
17.         self._tasks[task_id] = t; return t
18.     def unregister(self, task_id):
19.         t = self._tasks.pop(task_id, None)
20.         if t and t._task: t._task.cancel()
21.         return t is not None
22.     async def start(self):
23.         self._running = True
24.         for t in self._tasks.values(): t._task = asyncio.create_task(self._run_loop(t))
25.     async def stop(self):
26.         self._running = False
27.         for t in self._tasks.values():
28.             if t._task: t._task.cancel()
29.     async def _run_loop(self, s):
30.         while self._running:
31.             await asyncio.sleep(s.interval_seconds)
32.             try:
33.                 await s.func(); s.last_run = datetime.utcnow(); s.run_count += 1
34.             except asyncio.CancelledError: break
35.             except Exception as e: s.error_count += 1; logger.error("Task '%s' error: %s", s.task_id, e)
36.     def list_tasks(self):
37.         return [{"task_id": t.task_id, "description": t.description,
38.                  "interval_seconds": t.interval_seconds,
39.                  "last_run": t.last_run.isoformat() if t.last_run else None,
40.                  "run_count": t.run_count, "error_count": t.error_count}
41.                 for t in self._tasks.values()]
